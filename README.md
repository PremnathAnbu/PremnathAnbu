# GenAI + FastAPI — RAG endpoint example
from fastapi import FastAPI, UploadFile
from langchain.chains import RetrievalQA
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import FAISS

app = FastAPI(title="GenAI API 🤖", version="1.0")

@app.post("/rag/ask")
async def ask_document(query: str, file: UploadFile):
    """Upload a doc and ask questions about it."""
    docs     = load_and_split(file)                          # Parse & chunk
    vectordb = FAISS.from_documents(docs, OpenAIEmbeddings()) # Embed & index
    chain    = RetrievalQA.from_chain_type(                  # RAG chain
        llm=ChatOpenAI(model="gpt-4"),
        retriever=vectordb.as_retriever(search_kwargs={"k": 4})
    )
    return {"answer": chain.run(query)}

@app.post("/predict")
async def predict(features: InputSchema):
    """Traditional ML prediction endpoint."""
    prediction = model.predict(features.to_array())
    return {"prediction": int(prediction[0]), "confidence": 0.94}

@app.get("/health")
async def health(): return {"status": "✅ API is live"}
